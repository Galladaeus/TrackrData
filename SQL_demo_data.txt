-- This script is to insert a test dataset of visits

WITH RECURSIVE
days(day_number) AS (
    SELECT 2
    UNION ALL
    SELECT day_number + 1
    FROM days
    WHERE day_number < 15
),
visit_plan(
    location_id,
    pattern,
    local_start,
    duration_minutes,
    day_number
) AS (
    -- Oakland Hall: afternoon returns and overnight stays.
    SELECT 12, 'dorm-afternoon', '16:10:00',
           10 + (day_number % 2) * 5, day_number
    FROM days

    UNION ALL
    SELECT 12, 'dorm-overnight', '20:30:00',
           690 + (day_number % 3) * 15, day_number
    FROM days

    -- McKeldin: daily morning sessions plus six afternoon sessions.
    UNION ALL
    SELECT 7, 'library-morning', '09:00:00',
           165 + (day_number % 3) * 10, day_number
    FROM days

    UNION ALL
    SELECT 7, 'library-afternoon', '13:30:00',
           135 + (day_number % 2) * 15, day_number
    FROM days
    WHERE day_number <= 7

    -- STAMP: twelve lunch stops.
    UNION ALL
    SELECT 8, 'stamp-lunch', '12:10:00',
           35 + (day_number % 3) * 5, day_number
    FROM days
    WHERE day_number <= 13

    -- Shopping and grocery stops.
    UNION ALL
    SELECT 11, 'lidl', '16:30:00',
           25 + (day_number % 3) * 5, day_number
    FROM days
    WHERE day_number IN (2, 4, 6, 8, 10, 12)

    UNION ALL
    SELECT 9, 'target', '17:30:00',
           30 + (day_number % 2) * 10, day_number
    FROM days
    WHERE day_number IN (3, 6, 9, 12)

    UNION ALL
    SELECT 13, 'safeway', '18:30:00',
           20 + (day_number % 3) * 5, day_number
    FROM days
    WHERE day_number IN (5, 9, 13)

    UNION ALL
    SELECT 10, 'home-depot', '19:15:00',
           35 + (day_number % 2) * 10, day_number
    FROM days
    WHERE day_number IN (7, 14)
),
timed_visits AS (
    SELECT
        p.*,

        CAST(
            strftime(
                '%s',
                date(
                    'now',
                    'localtime',
                    '-' || day_number || ' days'
                ) || ' ' || local_start,
                'utc'
            ) AS INTEGER
        ) * 1000 AS arrival,

        'Synthetic demo visit [trackr-demo-v1:'
            || location_id || ':'
            || pattern || ':'
            || day_number || ']' AS seed_note

    FROM visit_plan AS p
)
INSERT INTO Visit (
    location_uid,
    name,
    arrived_at,
    departed_at,
    notes,
    manual_entry,
    accuracy_meters
)
SELECT
    l.uid,
    l.name,
    t.arrival,
    t.arrival + t.duration_minutes * 60000,
    t.seed_note,
    1,
    NULL
FROM timed_visits AS t
JOIN Location AS l
    ON l.uid = t.location_id
WHERE NOT EXISTS (
    SELECT 1
    FROM Visit AS existing
    WHERE existing.location_uid = l.uid
      AND existing.notes = t.seed_note
);
